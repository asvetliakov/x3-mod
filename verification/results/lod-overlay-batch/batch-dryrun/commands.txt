# Merged-LOD overlay batch mode: dry runs (2026-09-23; read-only against the bottle, no Wine, no --install).
# Run from the repository root. S = the session scratchpad; the bottle addon/05 pilot overlay is skipped through its
# (legacy) marker. Wall times are on this host with the given --jobs. The first (b) and (c) runs had the game closed (ps check); the b2 rerun overlapped the user's flight, so its timing is not clean.
ps -axww -o args= | grep '[X]3AP.exe'   # must print nothing before a batch run
# (a) host tests: the batch module plus the census, bob1 (incl. the existing lod_overlay tests) and game_guard
PYTHONPATH=verification/probe /usr/bin/python3 -m unittest verification.analysis.test_bob1 verification.analysis.test_lod_batch_census verification.analysis.test_lod_overlay_batch verification.analysis.test_game_guard
# (b) flown-sector bodies of the bottle (the --only list is sectors.txt itself: every 'N draws NAME' row of the three sectors)
/usr/bin/python3 tools/analysis/lod_overlay.py --batch --dry-run --only verification/results/lod-overlay-batch/sectors.txt --out $S/batch-b --jobs 6 > $S/batch-b.log 2>&1
cp $S/batch-b/x3m-lod-batch-summary.txt bottle-sectors-summary.txt
/usr/bin/python3 verification/results/lod-overlay-batch/batch-dryrun/record_digest.py $S/batch-b/x3m-lod-batch.json > bottle-sectors-record_out.txt   # per-body draws / sizes / ratios from the record JSON
# (c) synthetic vanilla+mod root (verification/results/lod-overlay-mods/commands.txt recipe: bottle 01..13 + addon/01..04,
#     /tmp/x3-mod1 addon/05..11, /tmp/x3-mod2 addon/12; M=$S/modroot), restricted to three mod-catalogue .bob bodies (only_mods.txt,
#     trimmed by hand from the nine mod_enumeration.py picked: a trailing-byte turret prop, a mixed-effects hull, a station)
/usr/bin/python3 tools/analysis/lod_overlay.py --batch --dry-run --game $S/modroot --only verification/results/lod-overlay-batch/batch-dryrun/only_mods.txt --out $S/batch-c --jobs 3 > $S/batch-c.log 2>&1
cp $S/batch-c/x3m-lod-batch-summary.txt modroot-summary.txt
/usr/bin/python3 verification/results/lod-overlay-batch/batch-dryrun/record_digest.py $S/batch-c/x3m-lod-batch.json > modroot-record_out.txt
# (c') census-only enumeration of the whole mod root with .bob members and text bodies (no baking; eligible counts by category; 4 jobs, ~4 min)
/usr/bin/python3 verification/results/lod-overlay-batch/batch-dryrun/mod_enumeration.py $S/modroot --jobs 4 > mod_enumeration_out.txt   # also writes only_mods.txt
